Integración en Next.js 15+ (CMS)

// app/herramientas/guias/page.tsx
import { Suspense } from 'react';
import { getSession } from '@/lib/auth';

export default async function GuidesPage() {
  const session = await getSession();
  const token = session?.accessToken;

  // Token se pasa via URL query param
  const editorUrl = new URL('https://guias.discoolver.com');
  editorUrl.searchParams.set('token', token);
  editorUrl.searchParams.set('city', 'Madrid'); // Opcional: prefill city

  return (
    <main className="h-screen">
      <div className="bg-gradient-to-r from-pink-600 to-orange-500 text-white p-6">
        <h1 className="text-3xl font-bold">Editor de Guías</h1>
        <p>Crea y edita guías de viaje con IA</p>
      </div>

      <Suspense fallback={<LoadingSpinner />}>
        <iframe
          src={editorUrl.toString()}
          className="w-full h-full border-0"
          title="Discoolver Guide Editor"
          allow="camera; microphone; clipboard-read; clipboard-write"
        />
      </Suspense>
    </main>
  );
}

Integración en React (con Vite)

// components/GuideEditor.jsx
import { useEffect, useRef } from 'react';
import { useAuth } from '@/contexts/AuthContext';

export function GuideEditor({ cityFilter = null }) {
  const { token } = useAuth();
  const iframeRef = useRef(null);

  useEffect(() => {
    if (!token) return;

    // Construir URL con token
    const editorUrl = new URL('https://guias.discoolver.com');
    editorUrl.searchParams.set('token', token);
    if (cityFilter) {
      editorUrl.searchParams.set('city', cityFilter);
    }

    // Cargar en iFrame
    if (iframeRef.current) {
      iframeRef.current.src = editorUrl.toString();
    }
  }, [token, cityFilter]);

  // Escuchar eventos del editor
  useEffect(() => {
    const handleMessage = (event) => {
      // Validar origen
      if (!event.origin.includes('discoolver.com')) return;

      switch (event.data.type) {
        case 'GUIDE_SAVED':
          console.log('Guía guardada:', event.data.guide);
          // Refrescar lista de guías
          break;
        case 'GUIDE_EXPORTED':
          console.log('PDF generado:', event.data.pdf_url);
          break;
        case 'DISCOOLVER_LOGOUT':
          // Redirigir a login
          window.location.href = '/login';
          break;
      }
    };

    window.addEventListener('message', handleMessage);
    return () => window.removeEventListener('message', handleMessage);
  }, []);

  return (
    <iframe
      ref={iframeRef}
      title="Discoolver Guide Editor"
      className="w-full h-screen border-0"
      allow="camera; microphone; clipboard-read; clipboard-write"
      sandbox="allow-same-origin allow-scripts allow-forms allow-popups allow-modals"
    />
  );
}

// Uso:
<GuideEditor cityFilter="Madrid" />

Manejo de Tokens

// Flujo de autenticación

// 1. CMS genera token
const session = await getSession(); // { accessToken, expiresAt, ... }

// 2. CMS pasa token via URL al editor
const editorUrl = `https://guias.discoolver.com?token=${session.accessToken}`;

// 3. Editor lee token de URL y lo guarda en sessionStorage
const params = new URLSearchParams(location.search);
const token = params.get('token'); // "Bearer abc123..."
sessionStorage.setItem('dg_editor_token', token);

// 4. Editor incluye token en todos los requests
fetch('https://api.discoolver.com/cms/v2/guides', {
  headers: {
    'Authorization': `Bearer ${token}`
  }
});

// 5. Si token expira (401):
if (response.status === 401) {
  // Editor muestra modal: "Sesión expirada"
  // Usuario debe reautenticarse en el CMS padre
  window.parent.postMessage(
    { type: 'DISCOOLVER_SESSION_EXPIRED' },
    'https://cms.discoolver.com'
  );
}

// ⚠️ NOTA: sessionStorage se borra al cerrar la pestaña
// No usar localStorage en iFrame (cross-origin issues)

🔒 Consideraciones de Seguridad

// SEGURIDAD EN iFrame

// 1. Token en URL vs Header
// ✅ Mejor: Pasar token via header (pero en iFrame cross-origin es difícil)
// ✅ OK: Token en URL con HTTPS + limpiar después de usar
// ❌ Nunca: Token en localStorage cross-domain

// 2. CORS y SameSite
// En Spring Boot, configurar:
@Configuration
@EnableWebMvc
public class CorsConfig implements WebMvcConfigurer {
  @Override
  public void addCorsMappings(CorsRegistry registry) {
    registry.addMapping("/cms/v2/**")
      .allowedOrigins(
        "https://cms.discoolver.com",
        "https://guias.discoolver.com"
      )
      .allowedMethods("GET", "POST", "PUT", "DELETE", "PATCH")
      .allowedHeaders("Authorization", "Content-Type")
      .allowCredentials(false)
      .maxAge(3600);
  }
}

// 3. CSP (Content Security Policy)
// En respuesta HTTP del editor:
Content-Security-Policy:
  default-src 'self';
  script-src 'self' 'unsafe-inline' https://cdn.jsdelivr.net;
  img-src 'self' https: data:;
  style-src 'self' 'unsafe-inline';
  connect-src 'self' https://api.discoolver.com;
  frame-ancestors https://cms.discoolver.com;

// 4. Sandbox attributes en HTML
<iframe
  sandbox="
    allow-same-origin
    allow-scripts
    allow-forms
    allow-popups
    allow-modals
  "
/>

// 5. Validar origen en postMessage
window.addEventListener('message', (event) => {
  // ✅ Validar origen del mensaje
  if (!event.origin.startsWith('https://cms.discoolver.com')) {
    return; // Ignorar
  }
  // Procesar evento seguro
});
